cmake_minimum_required(VERSION 3.21)
project(HiveMonitorDesktop VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network WebEngineWidgets)
qt_standard_project_setup()
qt_add_executable(HiveMonitorDesktop main.cpp)
target_link_libraries(HiveMonitorDesktop PRIVATE Qt6::Widgets Qt6::Network Qt6::WebEngineWidgets)
if(WIN32)
    set_target_properties(HiveMonitorDesktop PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
install(TARGETS HiveMonitorDesktop RUNTIME DESTINATION bin BUNDLE DESTINATION .)
